cmake_minimum_required(VERSION 3.21)
project(three_ds_live_qt VERSION 1.2.4 LANGUAGES C CXX)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
add_library(live_view STATIC viewer.cpp debug_window.cpp)
target_link_libraries(live_view PUBLIC Qt6::Widgets)
if(WIN32)
    target_link_libraries(live_view PRIVATE imm32)
endif()
target_compile_features(live_view PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(live_view PRIVATE /utf-8 /W4)
endif()
option(LIVE_NATIVE_APP "Build the native Windows receiver" ON)
if(LIVE_NATIVE_APP)
    find_package(Qt6 REQUIRED COMPONENTS Network)
    set(LIVE_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../build/native-deps" CACHE PATH "Extracted native dependencies")
    set(ORT "${LIVE_DEPS}/ort")
    set(CV "${LIVE_DEPS}/cv/opencv/build")
    set(AV "${LIVE_DEPS}/av/ffmpeg-9.0.2-full_build-shared")
    set(CODEC "${CMAKE_CURRENT_SOURCE_DIR}/../../codec")
    add_library(live_client STATIC client.cpp receiver.cpp wlan_stream.cpp transport.cpp diagnostics.cpp imaging.cpp native_detail.cpp timeline.cpp pipeline.cpp recording.cpp model_check.cpp
        "${CODEC}/source/live_audio.c" "${CODEC}/source/live_codec.c" "${CODEC}/host/live_audio_output.c")
    target_compile_features(live_client PRIVATE cxx_std_17)
    set_target_properties(live_client PROPERTIES AUTOMOC ON)
    target_compile_definitions(live_client PRIVATE LIVE_VERSION="${PROJECT_VERSION}" NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(live_client PRIVATE /utf-8 /W3 /bigobj)
    target_include_directories(live_client PRIVATE "${CODEC}/include" "${ORT}/build/native/include"
        "${LIVE_DEPS}/dml/include" "${CV}/include" "${AV}/include")
    target_link_libraries(live_client PUBLIC live_view Qt6::Network PRIVATE winmm wlanapi iphlpapi ws2_32
        "${ORT}/runtimes/win-x64/native/onnxruntime.lib" "${CV}/x64/vc16/lib/opencv_world4120.lib"
        "${AV}/lib/avformat.lib" "${AV}/lib/avcodec.lib" "${AV}/lib/avutil.lib" "${AV}/lib/swscale.lib")
    option(LIVE_TENSORRT "Use app-local FBCNN and RIFE TensorRT engines on compatible GPUs" OFF)
    if(LIVE_TENSORRT)
        find_path(TRT_INCLUDE NvInferRuntime.h HINTS "${LIVE_DEPS}/trt/include" "$ENV{TensorRT_ROOT}/include" REQUIRED)
        find_path(CUDA_INCLUDE cuda_runtime_api.h HINTS "$ENV{CUDA_PATH}/include" REQUIRED)
        find_path(CUDA_CRT_INCLUDE crt/host_defines.h HINTS "${CUDA_INCLUDE}" REQUIRED)
        find_library(CUDA_STATIC cudart_static HINTS "$ENV{CUDA_PATH}/lib/x64" REQUIRED)
        target_sources(live_client PRIVATE image_trt.cpp)
        target_include_directories(live_client SYSTEM PRIVATE "${TRT_INCLUDE}" "${CUDA_INCLUDE}" "${CUDA_CRT_INCLUDE}")
        target_compile_definitions(live_client PRIVATE LIVE_TENSORRT)
        target_link_libraries(live_client PRIVATE "${CUDA_STATIC}")
    endif()
    add_library(ThreeDSLiveClient SHARED main.cpp)
    qt_add_resources(ThreeDSLiveClient app_icon PREFIX "/" FILES assets/3ds-live.png)
    target_link_libraries(ThreeDSLiveClient PRIVATE live_client)
    set_target_properties(ThreeDSLiveClient PROPERTIES OUTPUT_NAME "3ds-live-client"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/runtime")
    add_executable(ThreeDSLive WIN32 launcher.cpp app.rc)
    target_compile_features(ThreeDSLive PRIVATE cxx_std_17)
    target_compile_options(ThreeDSLive PRIVATE /utf-8)
    target_link_libraries(ThreeDSLive PRIVATE user32)
    set_property(TARGET ThreeDSLive PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set_target_properties(ThreeDSLive PROPERTIES OUTPUT_NAME "3DS-Live")
    add_dependencies(ThreeDSLive ThreeDSLiveClient)
    install(TARGETS ThreeDSLive RUNTIME DESTINATION .)
    install(TARGETS ThreeDSLiveClient RUNTIME DESTINATION runtime)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test_viewer test_viewer.cpp)
    target_link_libraries(test_viewer PRIVATE live_view Qt6::Test)
    if(MSVC)
        target_compile_options(test_viewer PRIVATE /utf-8)
    endif()
    add_test(NAME viewer COMMAND test_viewer)
    if(LIVE_NATIVE_APP)
        add_executable(test_transport test_transport.cpp transport.cpp)
        target_compile_features(test_transport PRIVATE cxx_std_17)
        target_link_libraries(test_transport PRIVATE Qt6::Core)
        add_test(NAME transport COMMAND test_transport)
        add_executable(test_media test_media.cpp timeline.cpp diagnostics.cpp transport.cpp
            "${CODEC}/source/live_audio.c" "${CODEC}/source/live_codec.c" "${CODEC}/host/live_audio_output.c")
        set_target_properties(test_media PROPERTIES AUTOMOC ON)
        target_compile_features(test_media PRIVATE cxx_std_17)
        target_compile_options(test_media PRIVATE /utf-8)
        target_compile_definitions(test_media PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_include_directories(test_media PRIVATE "${CODEC}/include" "${AV}/include")
        target_link_libraries(test_media PRIVATE Qt6::Test Qt6::Gui winmm "${AV}/lib/avformat.lib"
            "${AV}/lib/avcodec.lib" "${AV}/lib/avutil.lib" "${AV}/lib/swscale.lib")
        add_test(NAME media COMMAND test_media)
        add_executable(test_session test_session.cpp)
        target_compile_options(test_session PRIVATE /utf-8)
        target_link_libraries(test_session PRIVATE live_client Qt6::Test)
        add_test(NAME desktop_session COMMAND test_session)

        add_test(NAME picture COMMAND test_picture)
        add_executable(test_flow test_flow.cpp)
        target_link_libraries(test_flow PRIVATE live_client)
        add_test(NAME flow_occlusion COMMAND test_flow)
        add_executable(bench_jpeg bench_jpeg.cpp)
        target_include_directories(bench_jpeg PRIVATE "${CODEC}/include")
        target_link_libraries(bench_jpeg PRIVATE live_client)
    endif()
endif()
